    /**
     * Aligns {@code size} up to the next multiple of {@code alignment} (must be a power of two).
     */
    private static long alignBufferSize(final long size, final long alignment) {
        return (size + alignment - 1) & ~(alignment - 1);
    }

    // -----------------------------------------------------------------------
    // SPIR-V shaders  (same bytecode as the C++ backend)
    // -----------------------------------------------------------------------

    /**
     * Converts an {@code int[]} SPIR-V word array into the {@link ByteBuffer} that
     * {@link VkShaderModuleCreateInfo#pCode(ByteBuffer)} expects (little-endian).
     */
    private static ByteBuffer intArrayToByteBuffer(final int[] words) {
        final ByteBuffer buf = MemoryUtil.memAlloc(words.length * Integer.BYTES);
        buf.order(java.nio.ByteOrder.LITTLE_ENDIAN);
        for (final int w : words) buf.putInt(w);
        buf.flip();
        return buf;
    }

    /**
     * Initialises the Vulkan backend. Must be called before {@link #renderDrawData}.
     *
     * @param info fully populated {@link InitInfo}
     * @return {@code true} on success
     */
    public boolean init(final InitInfo info) {
        final ImGuiIO io = ImGui.getIO();
        io.setBackendRendererName("imgui-java_impl_vulkan");
        io.addBackendFlags(ImGuiBackendFlags.RendererHasVtxOffset);

        data = new Data();
        data.initInfo = info;

        // Query nonCoherentAtomSize from physical device limits
        try (MemoryStack stack = MemoryStack.stackPush()) {
            VkPhysicalDeviceProperties props = VkPhysicalDeviceProperties.malloc(stack);
            vkGetPhysicalDeviceProperties(info.physicalDevice, props);
            data.nonCoherentAtomSize = props.limits().nonCoherentAtomSize();
        }

        createDeviceObjects();
        return true;
    }

    /**
     * Shuts down the backend and releases all Vulkan resources.
     * The caller must ensure the GPU has finished before calling this.
     */
    public void shutdown() {
        final ImGuiIO io = ImGui.getIO();
        destroyDeviceObjects();
        io.setBackendRendererName(null);
        data = null;
    }

    /**
     * Records ImGui draw commands into {@code commandBuffer}.
     * Must be called inside a render pass.
     *
     * @param drawData      the draw data from {@code ImGui.getDrawData()}
     * @param commandBuffer an active Vulkan command buffer inside the configured render pass
     */
    public void renderDrawData(final ImDrawData drawData, final VkCommandBuffer commandBuffer) {
        final int fbWidth = (int) (drawData.getDisplaySizeX() * drawData.getFramebufferScaleX());
        final int fbHeight = (int) (drawData.getDisplaySizeY() * drawData.getFramebufferScaleY());
        if (fbWidth <= 0 || fbHeight <= 0) return;

        // Lazily create font texture on first render
        if (data.fontDescSet == VK_NULL_HANDLE) {
            createFontsTexture(commandBuffer);
        }

        final InitInfo v = data.initInfo;

        // Cycle through per-frame render buffers
        if (data.vtxBuffers == null) {
            data.wrb_index = 0;
            data.wrb_count = v.imageCount;
            data.vtxBuffers = new long[v.imageCount];
            data.vtxBufferMemories = new long[v.imageCount];
            data.vtxBufferSizes = new long[v.imageCount];
            data.idxBuffers = new long[v.imageCount];
            data.idxBufferMemories = new long[v.imageCount];
            data.idxBufferSizes = new long[v.imageCount];
        }
        data.wrb_index = (data.wrb_index + 1) % data.wrb_count;
        final int fi = data.wrb_index;

        if (drawData.getTotalVtxCount() > 0) {
            final long vtxSize = alignBufferSize(
                    (long) drawData.getTotalVtxCount() * ImDrawData.sizeOfImDrawVert(), data.bufferMemoryAlignment);
            final long idxSize = alignBufferSize(
                    (long) drawData.getTotalIdxCount() * ImDrawData.sizeOfImDrawIdx(), data.bufferMemoryAlignment);

            if (data.vtxBuffers[fi] == VK_NULL_HANDLE || data.vtxBufferSizes[fi] < vtxSize) {
                destroyBuffer(data.vtxBuffers[fi], data.vtxBufferMemories[fi]);
                final long[] out = createBuffer(vtxSize, VK_BUFFER_USAGE_VERTEX_BUFFER_BIT);
                data.vtxBuffers[fi] = out[0];
                data.vtxBufferMemories[fi] = out[1];
                data.vtxBufferSizes[fi] = vtxSize;
            }
            if (data.idxBuffers[fi] == VK_NULL_HANDLE || data.idxBufferSizes[fi] < idxSize) {
                destroyBuffer(data.idxBuffers[fi], data.idxBufferMemories[fi]);
                final long[] out = createBuffer(idxSize, VK_BUFFER_USAGE_INDEX_BUFFER_BIT);
                data.idxBuffers[fi] = out[0];
                data.idxBufferMemories[fi] = out[1];
                data.idxBufferSizes[fi] = idxSize;
            }

            // Upload vertex and index data
            uploadMeshData(drawData, fi, vtxSize, idxSize);
        }

        setupRenderState(drawData, commandBuffer, fi, fbWidth, fbHeight);

        // Render command lists
        final float clipOffX = drawData.getDisplayPosX();
        final float clipOffY = drawData.getDisplayPosY();
        final float clipScaleX = drawData.getFramebufferScaleX();
        final float clipScaleY = drawData.getFramebufferScaleY();

        long lastDescSet = VK_NULL_HANDLE;
        int globalVtxOffset = 0;
        int globalIdxOffset = 0;

        try (MemoryStack stack = MemoryStack.stackPush()) {
            final LongBuffer descSetBuf = stack.mallocLong(1);
            final ImVec4 clipRect = new ImVec4();

            for (int n = 0; n < drawData.getCmdListsCount(); n++) {
                final int cmdCount = drawData.getCmdListCmdBufferSize(n);
                for (int cmd = 0; cmd < cmdCount; cmd++) {
                    // User callbacks are not handled here – extend this class if needed.
                    if (drawData.getCmdListCmdBufferUserCallbackData(n, cmd) != 0) {
                        continue;
                    }

                    drawData.getCmdListCmdBufferClipRect(clipRect, n, cmd);
                    final float clipMinX = Math.max((clipRect.x - clipOffX) * clipScaleX, 0.0f);
                    final float clipMinY = Math.max((clipRect.y - clipOffY) * clipScaleY, 0.0f);
                    final float clipMaxX = Math.min((clipRect.z - clipOffX) * clipScaleX, fbWidth);
                    final float clipMaxY = Math.min((clipRect.w - clipOffY) * clipScaleY, fbHeight);
                    if (clipMaxX <= clipMinX || clipMaxY <= clipMinY) continue;

                    // Scissor
                    final VkRect2D.Buffer scissor = VkRect2D.malloc(1, stack);
                    scissor.get(0).offset().set((int) clipMinX, (int) clipMinY);
                    scissor.get(0).extent().set(
                            (int) (clipMaxX - clipMinX),
                            (int) (clipMaxY - clipMinY));
                    vkCmdSetScissor(commandBuffer, 0, scissor);

                    // Bind texture descriptor set
                    final long texId = drawData.getCmdListCmdBufferTextureId(n, cmd);
                    final long descSet = (texId != 0) ? texId : data.fontDescSet;
                    if (descSet != lastDescSet) {
                        descSetBuf.put(0, descSet);
                        vkCmdBindDescriptorSets(commandBuffer,
                                VK_PIPELINE_BIND_POINT_GRAPHICS,
                                data.pipelineLayout,
                                0, descSetBuf, null);
                        lastDescSet = descSet;
                    }

                    final int elemCount = drawData.getCmdListCmdBufferElemCount(n, cmd);
                    final int idxOffset = drawData.getCmdListCmdBufferIdxOffset(n, cmd);
                    final int vtxOffset = drawData.getCmdListCmdBufferVtxOffset(n, cmd);
                    vkCmdDrawIndexed(commandBuffer, elemCount, 1,
                            idxOffset + globalIdxOffset,
                            vtxOffset + globalVtxOffset, 0);
                }
                globalIdxOffset += drawData.getCmdListIdxBufferSize(n);
                globalVtxOffset += drawData.getCmdListVtxBufferSize(n);
            }

            // Restore full scissor at the end so we don't pollute subsequent draw calls
            final VkRect2D.Buffer fullScissor = VkRect2D.malloc(1, stack);
            fullScissor.get(0).offset().set(0, 0);
            fullScissor.get(0).extent().set(fbWidth, fbHeight);
            vkCmdSetScissor(commandBuffer, 0, fullScissor);
        }
    }

    /**
     * Registers an arbitrary {@code VkImageView} as an imgui texture and returns
     * the {@code VkDescriptorSet} handle to use as {@code ImTextureID}.
     *
     * @param sampler     the sampler to bind
     * @param imageView   the image view
     * @param imageLayout the current image layout (usually {@code VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL})
     * @return a {@code VkDescriptorSet} handle usable as {@code ImTextureID}
     */
    public long addTexture(final long sampler, final long imageView, final int imageLayout) {
        final InitInfo v = data.initInfo;
        try (MemoryStack stack = MemoryStack.stackPush()) {
            // Allocate descriptor set
            final LongBuffer setLayouts = stack.longs(data.descriptorSetLayout);
            final VkDescriptorSetAllocateInfo allocInfo = VkDescriptorSetAllocateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_DESCRIPTOR_SET_ALLOCATE_INFO)
                    .descriptorPool(v.descriptorPool)
                    .pSetLayouts(setLayouts);

            final LongBuffer pDescSet = stack.mallocLong(1);
            checkVk(vkAllocateDescriptorSets(v.device, allocInfo, pDescSet));
            final long descSet = pDescSet.get(0);

            // Update descriptor
            final VkDescriptorImageInfo.Buffer imageInfo = VkDescriptorImageInfo.calloc(1, stack)
                    .sampler(sampler)
                    .imageView(imageView)
                    .imageLayout(imageLayout);

            final VkWriteDescriptorSet.Buffer write = VkWriteDescriptorSet.calloc(1, stack)
                    .sType(VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET)
                    .dstSet(descSet)
                    .descriptorCount(1)
                    .descriptorType(VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER)
                    .pImageInfo(imageInfo);

            vkUpdateDescriptorSets(v.device, write, null);
            return descSet;
        }
    }

    /**
     * Frees a descriptor set previously allocated by {@link #addTexture}.
     *
     * @param descSet the descriptor set to free
     */
    public void removeTexture(final long descSet) {
        vkFreeDescriptorSets(data.initInfo.device, data.initInfo.descriptorPool, descSet);
    }

    protected void createDeviceObjects() {
        final InitInfo v = data.initInfo;
        try (MemoryStack stack = MemoryStack.stackPush()) {

            // ---- Descriptor set layout (combined image sampler, set=0, binding=0) ----
            final VkDescriptorSetLayoutBinding.Buffer binding = VkDescriptorSetLayoutBinding.calloc(1, stack)
                    .binding(0)
                    .descriptorType(VK_DESCRIPTOR_TYPE_COMBINED_IMAGE_SAMPLER)
                    .descriptorCount(1)
                    .stageFlags(VK_SHADER_STAGE_FRAGMENT_BIT);

            final VkDescriptorSetLayoutCreateInfo dslInfo = VkDescriptorSetLayoutCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_DESCRIPTOR_SET_LAYOUT_CREATE_INFO)
                    .pBindings(binding);

            final LongBuffer pDsl = stack.mallocLong(1);
            checkVk(vkCreateDescriptorSetLayout(v.device, dslInfo, v.allocator, pDsl));
            data.descriptorSetLayout = pDsl.get(0);

            // ---- Font sampler ----
            final VkSamplerCreateInfo samplerInfo = VkSamplerCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_SAMPLER_CREATE_INFO)
                    .magFilter(VK_FILTER_LINEAR)
                    .minFilter(VK_FILTER_LINEAR)
                    .mipmapMode(VK_SAMPLER_MIPMAP_MODE_LINEAR)
                    .addressModeU(VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE)
                    .addressModeV(VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE)
                    .addressModeW(VK_SAMPLER_ADDRESS_MODE_CLAMP_TO_EDGE)
                    .minLod(-1000f)
                    .maxLod(1000f)
                    .maxAnisotropy(1.0f);

            final LongBuffer pSampler = stack.mallocLong(1);
            checkVk(vkCreateSampler(v.device, samplerInfo, v.allocator, pSampler));
            data.fontSampler = pSampler.get(0);

            // ---- Pipeline layout (push constants: vec2 scale + vec2 translate = 4 floats) ----
            final VkPushConstantRange.Buffer pcRange = VkPushConstantRange.calloc(1, stack)
                    .stageFlags(VK_SHADER_STAGE_VERTEX_BIT)
                    .offset(0)
                    .size(Float.BYTES * 4);

            final LongBuffer pDsl2 = stack.longs(data.descriptorSetLayout);
            final VkPipelineLayoutCreateInfo layoutInfo = VkPipelineLayoutCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO)
                    .pSetLayouts(pDsl2)
                    .pPushConstantRanges(pcRange);

            final LongBuffer pPipelineLayout = stack.mallocLong(1);
            checkVk(vkCreatePipelineLayout(v.device, layoutInfo, v.allocator, pPipelineLayout));
            data.pipelineLayout = pPipelineLayout.get(0);

            // ---- Shader modules ----
            data.shaderModuleVert = createShaderModule(intArrayToByteBuffer(VERT_SPV));
            data.shaderModuleFrag = createShaderModule(intArrayToByteBuffer(FRAG_SPV));

            // ---- Graphics pipeline ----
            data.pipeline = createPipeline();

            // ---- Texture upload command pool + buffer ----
            final VkCommandPoolCreateInfo poolInfo = VkCommandPoolCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_COMMAND_POOL_CREATE_INFO)
                    .queueFamilyIndex(v.queueFamily);

            final LongBuffer pPool = stack.mallocLong(1);
            checkVk(vkCreateCommandPool(v.device, poolInfo, v.allocator, pPool));
            data.texCommandPool = pPool.get(0);

            final VkCommandBufferAllocateInfo cbAllocInfo = VkCommandBufferAllocateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_COMMAND_BUFFER_ALLOCATE_INFO)
                    .commandPool(data.texCommandPool)
                    .level(VK_COMMAND_BUFFER_LEVEL_PRIMARY)
                    .commandBufferCount(1);

            final PointerBuffer pCB = stack.mallocPointer(1);
            checkVk(vkAllocateCommandBuffers(v.device, cbAllocInfo, pCB));
            data.texCommandBuffer = pCB.get(0);
        }
    }

    // -----------------------------------------------------------------------
    // Device object creation / destruction
    // -----------------------------------------------------------------------

    protected void destroyDeviceObjects() {
        final InitInfo v = data.initInfo;

        // Destroy per-frame render buffers
        if (data.vtxBuffers != null) {
            for (int i = 0; i < data.wrb_count; i++) {
                destroyBuffer(data.vtxBuffers[i], data.vtxBufferMemories[i]);
                destroyBuffer(data.idxBuffers[i], data.idxBufferMemories[i]);
            }
        }

        // Destroy font texture
        destroyFontsTexture();

        if (data.texCommandPool != VK_NULL_HANDLE) {
            vkDestroyCommandPool(v.device, data.texCommandPool, v.allocator);
            data.texCommandPool = VK_NULL_HANDLE;
            data.texCommandBuffer = VK_NULL_HANDLE;
        }
        if (data.fontSampler != VK_NULL_HANDLE) {
            vkDestroySampler(v.device, data.fontSampler, v.allocator);
            data.fontSampler = VK_NULL_HANDLE;
        }
        if (data.pipeline != VK_NULL_HANDLE) {
            vkDestroyPipeline(v.device, data.pipeline, v.allocator);
            data.pipeline = VK_NULL_HANDLE;
        }
        if (data.pipelineLayout != VK_NULL_HANDLE) {
            vkDestroyPipelineLayout(v.device, data.pipelineLayout, v.allocator);
            data.pipelineLayout = VK_NULL_HANDLE;
        }
        if (data.shaderModuleVert != VK_NULL_HANDLE) {
            vkDestroyShaderModule(v.device, data.shaderModuleVert, v.allocator);
            data.shaderModuleVert = VK_NULL_HANDLE;
        }
        if (data.shaderModuleFrag != VK_NULL_HANDLE) {
            vkDestroyShaderModule(v.device, data.shaderModuleFrag, v.allocator);
            data.shaderModuleFrag = VK_NULL_HANDLE;
        }
        if (data.descriptorSetLayout != VK_NULL_HANDLE) {
            vkDestroyDescriptorSetLayout(v.device, data.descriptorSetLayout, v.allocator);
            data.descriptorSetLayout = VK_NULL_HANDLE;
        }
    }

    /**
     * Creates the font texture and uploads it to the GPU.
     * The command buffer must be in a recording state so we can issue the transfer.
     * After this method returns the caller must submit and wait idle before rendering.
     */
    public void createFontsTexture(final VkCommandBuffer commandBuffer) {
        final InitInfo v = data.initInfo;
        final ImFontAtlas fontAtlas = ImGui.getIO().getFonts();

        // Retrieve RGBA32 pixel data
        final ImInt width = new ImInt();
        final ImInt height = new ImInt();
        final ByteBuffer pixels = fontAtlas.getTexDataAsRGBA32(width, height);
        final int w = width.get();
        final int h = height.get();
        final long uploadSize = (long) w * h * 4;

        try (MemoryStack stack = MemoryStack.stackPush()) {

            // ---- Create device-local image ----
            final VkImageCreateInfo imageInfo = VkImageCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_IMAGE_CREATE_INFO)
                    .imageType(VK_IMAGE_TYPE_2D)
                    .format(VK_FORMAT_R8G8B8A8_UNORM)
                    .mipLevels(1)
                    .arrayLayers(1)
                    .samples(VK_SAMPLE_COUNT_1_BIT)
                    .tiling(VK_IMAGE_TILING_OPTIMAL)
                    .usage(VK_IMAGE_USAGE_SAMPLED_BIT | VK_IMAGE_USAGE_TRANSFER_DST_BIT)
                    .sharingMode(VK_SHARING_MODE_EXCLUSIVE)
                    .initialLayout(VK_IMAGE_LAYOUT_UNDEFINED);
            imageInfo.extent().set(w, h, 1);

            final LongBuffer pImage = stack.mallocLong(1);
            checkVk(vkCreateImage(v.device, imageInfo, v.allocator, pImage));
            data.fontImage = pImage.get(0);

            final VkMemoryRequirements req = VkMemoryRequirements.malloc(stack);
            vkGetImageMemoryRequirements(v.device, data.fontImage, req);

            final VkMemoryAllocateInfo allocInfo = VkMemoryAllocateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO)
                    .allocationSize(Math.max(v.minAllocationSize, req.size()))
                    .memoryTypeIndex(findMemoryType(req.memoryTypeBits(), VK_MEMORY_PROPERTY_DEVICE_LOCAL_BIT));

            final LongBuffer pMem = stack.mallocLong(1);
            checkVk(vkAllocateMemory(v.device, allocInfo, v.allocator, pMem));
            data.fontMemory = pMem.get(0);
            checkVk(vkBindImageMemory(v.device, data.fontImage, data.fontMemory, 0));

            // ---- Image view ----
            final VkImageViewCreateInfo viewInfo = VkImageViewCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_IMAGE_VIEW_CREATE_INFO)
                    .image(data.fontImage)
                    .viewType(VK_IMAGE_VIEW_TYPE_2D)
                    .format(VK_FORMAT_R8G8B8A8_UNORM);
            viewInfo.subresourceRange()
                    .aspectMask(VK_IMAGE_ASPECT_COLOR_BIT)
                    .levelCount(1)
                    .layerCount(1);

            final LongBuffer pView = stack.mallocLong(1);
            checkVk(vkCreateImageView(v.device, viewInfo, v.allocator, pView));
            data.fontImageView = pView.get(0);

            // ---- Upload buffer ----
            final long alignedUploadSize = alignBufferSize(uploadSize, data.nonCoherentAtomSize);
            final VkBufferCreateInfo bufInfo = VkBufferCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_BUFFER_CREATE_INFO)
                    .size(alignedUploadSize)
                    .usage(VK_BUFFER_USAGE_TRANSFER_SRC_BIT)
                    .sharingMode(VK_SHARING_MODE_EXCLUSIVE);

            final LongBuffer pBuf = stack.mallocLong(1);
            checkVk(vkCreateBuffer(v.device, bufInfo, v.allocator, pBuf));
            data.fontUploadBuffer = pBuf.get(0);

            final VkMemoryRequirements bufReq = VkMemoryRequirements.malloc(stack);
            vkGetBufferMemoryRequirements(v.device, data.fontUploadBuffer, bufReq);
            data.bufferMemoryAlignment = Math.max(data.bufferMemoryAlignment, bufReq.alignment());

            final VkMemoryAllocateInfo bufAlloc = VkMemoryAllocateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO)
                    .allocationSize(Math.max(v.minAllocationSize, bufReq.size()))
                    .memoryTypeIndex(findMemoryType(bufReq.memoryTypeBits(),
                            VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT));

            final LongBuffer pBufMem = stack.mallocLong(1);
            checkVk(vkAllocateMemory(v.device, bufAlloc, v.allocator, pBufMem));
            data.fontUploadBufferMemory = pBufMem.get(0);
            checkVk(vkBindBufferMemory(v.device, data.fontUploadBuffer, data.fontUploadBufferMemory, 0));

            // ---- Map + copy pixels ----
            final PointerBuffer ppData = stack.mallocPointer(1);
            checkVk(vkMapMemory(v.device, data.fontUploadBufferMemory, 0, alignedUploadSize, 0, ppData));
            final ByteBuffer mapped = MemoryUtil.memByteBuffer(ppData.get(0), (int) alignedUploadSize);
            MemoryUtil.memCopy(MemoryUtil.memAddress(pixels), MemoryUtil.memAddress(mapped), uploadSize);

            final VkMappedMemoryRange.Buffer flushRange = VkMappedMemoryRange.calloc(1, stack)
                    .sType(VK_STRUCTURE_TYPE_MAPPED_MEMORY_RANGE)
                    .memory(data.fontUploadBufferMemory)
                    .size(alignedUploadSize);
            checkVk(vkFlushMappedMemoryRanges(v.device, flushRange));
            vkUnmapMemory(v.device, data.fontUploadBufferMemory);

            // ---- Record copy commands ----
            // Barrier: UNDEFINED → TRANSFER_DST
            final VkImageMemoryBarrier.Buffer copyBarrier = VkImageMemoryBarrier.calloc(1, stack)
                    .sType(VK_STRUCTURE_TYPE_IMAGE_MEMORY_BARRIER)
                    .dstAccessMask(VK_ACCESS_TRANSFER_WRITE_BIT)
                    .oldLayout(VK_IMAGE_LAYOUT_UNDEFINED)
                    .newLayout(VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL)
                    .srcQueueFamilyIndex(VK_QUEUE_FAMILY_IGNORED)
                    .dstQueueFamilyIndex(VK_QUEUE_FAMILY_IGNORED)
                    .image(data.fontImage);
            copyBarrier.get(0).subresourceRange()
                    .aspectMask(VK_IMAGE_ASPECT_COLOR_BIT)
                    .levelCount(1)
                    .layerCount(1);
            vkCmdPipelineBarrier(commandBuffer,
                    VK_PIPELINE_STAGE_HOST_BIT, VK_PIPELINE_STAGE_TRANSFER_BIT,
                    0, null, null, copyBarrier);

            // Copy buffer → image
            final VkBufferImageCopy.Buffer region = VkBufferImageCopy.calloc(1, stack);
            region.get(0).imageSubresource()
                    .aspectMask(VK_IMAGE_ASPECT_COLOR_BIT)
                    .layerCount(1);
            region.get(0).imageExtent().set(w, h, 1);
            vkCmdCopyBufferToImage(commandBuffer,
                    data.fontUploadBuffer, data.fontImage,
                    VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL, region);

            // Barrier: TRANSFER_DST → SHADER_READ
            final VkImageMemoryBarrier.Buffer useBarrier = VkImageMemoryBarrier.calloc(1, stack)
                    .sType(VK_STRUCTURE_TYPE_IMAGE_MEMORY_BARRIER)
                    .srcAccessMask(VK_ACCESS_TRANSFER_WRITE_BIT)
                    .dstAccessMask(VK_ACCESS_SHADER_READ_BIT)
                    .oldLayout(VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL)
                    .newLayout(VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL)
                    .srcQueueFamilyIndex(VK_QUEUE_FAMILY_IGNORED)
                    .dstQueueFamilyIndex(VK_QUEUE_FAMILY_IGNORED)
                    .image(data.fontImage);
            useBarrier.get(0).subresourceRange()
                    .aspectMask(VK_IMAGE_ASPECT_COLOR_BIT)
                    .levelCount(1)
                    .layerCount(1);
            vkCmdPipelineBarrier(commandBuffer,
                    VK_PIPELINE_STAGE_TRANSFER_BIT, VK_PIPELINE_STAGE_FRAGMENT_SHADER_BIT,
                    0, null, null, useBarrier);
        }

        // Allocate and write descriptor set for the font texture
        data.fontDescSet = addTexture(data.fontSampler, data.fontImageView,
                VK_IMAGE_LAYOUT_SHADER_READ_ONLY_OPTIMAL);

        // Store the descriptor set as the font atlas texture ID
        fontAtlas.setTexID(data.fontDescSet);
    }

    /**
     * Frees the font texture GPU resources.
     */
    public void destroyFontsTexture() {
        final InitInfo v = data.initInfo;
        if (data.fontDescSet != VK_NULL_HANDLE) {
            vkFreeDescriptorSets(v.device, v.descriptorPool, data.fontDescSet);
            data.fontDescSet = VK_NULL_HANDLE;
        }
        if (data.fontImageView != VK_NULL_HANDLE) {
            vkDestroyImageView(v.device, data.fontImageView, v.allocator);
            data.fontImageView = VK_NULL_HANDLE;
        }
        if (data.fontImage != VK_NULL_HANDLE) {
            vkDestroyImage(v.device, data.fontImage, v.allocator);
            data.fontImage = VK_NULL_HANDLE;
        }
        if (data.fontMemory != VK_NULL_HANDLE) {
            vkFreeMemory(v.device, data.fontMemory, v.allocator);
            data.fontMemory = VK_NULL_HANDLE;
        }
        if (data.fontUploadBuffer != VK_NULL_HANDLE) {
            vkDestroyBuffer(v.device, data.fontUploadBuffer, v.allocator);
            data.fontUploadBuffer = VK_NULL_HANDLE;
        }
        if (data.fontUploadBufferMemory != VK_NULL_HANDLE) {
            vkFreeMemory(v.device, data.fontUploadBufferMemory, v.allocator);
            data.fontUploadBufferMemory = VK_NULL_HANDLE;
        }
    }

    private void setupRenderState(
            final ImDrawData drawData,
            final VkCommandBuffer commandBuffer,
            final int fi,
            final int fbWidth,
            final int fbHeight) {

        try (MemoryStack stack = MemoryStack.stackPush()) {
            vkCmdBindPipeline(commandBuffer, VK_PIPELINE_BIND_POINT_GRAPHICS, data.pipeline);

            if (drawData.getTotalVtxCount() > 0) {
                final LongBuffer vtxBufs = stack.longs(data.vtxBuffers[fi]);
                final LongBuffer vtxOffset = stack.longs(0L);
                vkCmdBindVertexBuffers(commandBuffer, 0, vtxBufs, vtxOffset);
                final int idxType = (ImDrawData.sizeOfImDrawIdx() == 2)
                        ? VK_INDEX_TYPE_UINT16 : VK_INDEX_TYPE_UINT32;
                vkCmdBindIndexBuffer(commandBuffer, data.idxBuffers[fi], 0, idxType);
            }

            // Viewport
            final VkViewport.Buffer viewport = VkViewport.calloc(1, stack)
                    .x(0).y(0)
                    .width(fbWidth).height(fbHeight)
                    .minDepth(0.0f).maxDepth(1.0f);
            vkCmdSetViewport(commandBuffer, 0, viewport);

            // Push constants: vec2 scale, vec2 translate
            final FloatBuffer pc = stack.mallocFloat(4);
            pc.put(0, 2.0f / drawData.getDisplaySizeX());
            pc.put(1, 2.0f / drawData.getDisplaySizeY());
            pc.put(2, -1.0f - drawData.getDisplayPosX() * pc.get(0));
            pc.put(3, -1.0f - drawData.getDisplayPosY() * pc.get(1));
            vkCmdPushConstants(commandBuffer, data.pipelineLayout,
                    VK_SHADER_STAGE_VERTEX_BIT, 0, pc);
        }
    }

    /**
     * Uploads vertex and index data for frame index {@code fi}.
     */
    private void uploadMeshData(
            final ImDrawData drawData,
            final int fi,
            final long vtxSize,
            final long idxSize) {

        final InitInfo v = data.initInfo;
        try (MemoryStack stack = MemoryStack.stackPush()) {
            final PointerBuffer ppVtx = stack.mallocPointer(1);
            final PointerBuffer ppIdx = stack.mallocPointer(1);
            checkVk(vkMapMemory(v.device, data.vtxBufferMemories[fi], 0, vtxSize, 0, ppVtx));
            checkVk(vkMapMemory(v.device, data.idxBufferMemories[fi], 0, idxSize, 0, ppIdx));

            long vtxDst = ppVtx.get(0);
            long idxDst = ppIdx.get(0);
            final int vtxStride = ImDrawData.sizeOfImDrawVert();
            final int idxStride = ImDrawData.sizeOfImDrawIdx();

            for (int n = 0; n < drawData.getCmdListsCount(); n++) {
                final ByteBuffer vtxData = drawData.getCmdListVtxBufferData(n);
                final ByteBuffer idxData = drawData.getCmdListIdxBufferData(n);
                final long vtxBytes = (long) drawData.getCmdListVtxBufferSize(n) * vtxStride;
                final long idxBytes = (long) drawData.getCmdListIdxBufferSize(n) * idxStride;
                MemoryUtil.memCopy(MemoryUtil.memAddress(vtxData), vtxDst, vtxBytes);
                MemoryUtil.memCopy(MemoryUtil.memAddress(idxData), idxDst, idxBytes);
                vtxDst += vtxBytes;
                idxDst += idxBytes;
            }

            final VkMappedMemoryRange.Buffer ranges = VkMappedMemoryRange.calloc(2, stack);
            ranges.get(0).sType(VK_STRUCTURE_TYPE_MAPPED_MEMORY_RANGE)
                    .memory(data.vtxBufferMemories[fi])
                    .size(VK_WHOLE_SIZE);
            ranges.get(1).sType(VK_STRUCTURE_TYPE_MAPPED_MEMORY_RANGE)
                    .memory(data.idxBufferMemories[fi])
                    .size(VK_WHOLE_SIZE);
            checkVk(vkFlushMappedMemoryRanges(v.device, ranges));

            vkUnmapMemory(v.device, data.vtxBufferMemories[fi]);
            vkUnmapMemory(v.device, data.idxBufferMemories[fi]);
        }
    }

    /**
     * Creates a host-visible buffer.
     *
     * @param size  byte size
     * @param usage Vulkan buffer usage flags
     * @return {@code [bufferHandle, memoryHandle]}
     */
    private long[] createBuffer(final long size, final int usage) {
        final InitInfo v = data.initInfo;
        try (MemoryStack stack = MemoryStack.stackPush()) {
            final long alignedSize = alignBufferSize(Math.max(v.minAllocationSize, size), data.bufferMemoryAlignment);

            final VkBufferCreateInfo info = VkBufferCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_BUFFER_CREATE_INFO)
                    .size(alignedSize)
                    .usage(usage)
                    .sharingMode(VK_SHARING_MODE_EXCLUSIVE);

            final LongBuffer pBuf = stack.mallocLong(1);
            checkVk(vkCreateBuffer(v.device, info, v.allocator, pBuf));
            final long buf = pBuf.get(0);

            final VkMemoryRequirements req = VkMemoryRequirements.malloc(stack);
            vkGetBufferMemoryRequirements(v.device, buf, req);
            data.bufferMemoryAlignment = Math.max(data.bufferMemoryAlignment, req.alignment());

            final VkMemoryAllocateInfo alloc = VkMemoryAllocateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_MEMORY_ALLOCATE_INFO)
                    .allocationSize(req.size())
                    .memoryTypeIndex(findMemoryType(req.memoryTypeBits(), VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT));

            final LongBuffer pMem = stack.mallocLong(1);
            checkVk(vkAllocateMemory(v.device, alloc, v.allocator, pMem));
            final long mem = pMem.get(0);
            checkVk(vkBindBufferMemory(v.device, buf, mem, 0));
            return new long[]{buf, mem};
        }
    }

    private void destroyBuffer(final long buf, final long mem) {
        final InitInfo v = data.initInfo;
        if (buf != VK_NULL_HANDLE) vkDestroyBuffer(v.device, buf, v.allocator);
        if (mem != VK_NULL_HANDLE) vkFreeMemory(v.device, mem, v.allocator);
    }

    /**
     * Creates the graphics pipeline used for ImGui rendering.
     */
    private long createPipeline() {
        final InitInfo v = data.initInfo;
        try (MemoryStack stack = MemoryStack.stackPush()) {

            // Shader stages
            final ByteBuffer entryPoint = stack.UTF8("main");
            final VkPipelineShaderStageCreateInfo.Buffer stages = VkPipelineShaderStageCreateInfo.calloc(2, stack);
            stages.get(0)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO)
                    .stage(VK_SHADER_STAGE_VERTEX_BIT)
                    .module(data.shaderModuleVert)
                    .pName(entryPoint);
            stages.get(1)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO)
                    .stage(VK_SHADER_STAGE_FRAGMENT_BIT)
                    .module(data.shaderModuleFrag)
                    .pName(entryPoint);

            // Vertex input: pos(vec2) uv(vec2) col(rgba8)
            //   stride = sizeof(ImDrawVert) = 20 bytes
            final int stride = ImDrawData.sizeOfImDrawVert();
            final VkVertexInputBindingDescription.Buffer bindingDesc = VkVertexInputBindingDescription.calloc(1, stack)
                    .binding(0)
                    .stride(stride)
                    .inputRate(VK_VERTEX_INPUT_RATE_VERTEX);

            final VkVertexInputAttributeDescription.Buffer attrDesc = VkVertexInputAttributeDescription.calloc(3, stack);
            attrDesc.get(0).location(0).binding(0).format(VK_FORMAT_R32G32_SFLOAT).offset(0);    // pos
            attrDesc.get(1).location(1).binding(0).format(VK_FORMAT_R32G32_SFLOAT).offset(8);    // uv
            attrDesc.get(2).location(2).binding(0).format(VK_FORMAT_R8G8B8A8_UNORM).offset(16);  // col

            final VkPipelineVertexInputStateCreateInfo vertexInput = VkPipelineVertexInputStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_VERTEX_INPUT_STATE_CREATE_INFO)
                    .pVertexBindingDescriptions(bindingDesc)
                    .pVertexAttributeDescriptions(attrDesc);

            final VkPipelineInputAssemblyStateCreateInfo inputAssembly = VkPipelineInputAssemblyStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_INPUT_ASSEMBLY_STATE_CREATE_INFO)
                    .topology(VK_PRIMITIVE_TOPOLOGY_TRIANGLE_LIST);

            final VkPipelineViewportStateCreateInfo viewportState = VkPipelineViewportStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_VIEWPORT_STATE_CREATE_INFO)
                    .viewportCount(1)
                    .scissorCount(1);

            final VkPipelineRasterizationStateCreateInfo rasterState = VkPipelineRasterizationStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_RASTERIZATION_STATE_CREATE_INFO)
                    .polygonMode(VK_POLYGON_MODE_FILL)
                    .cullMode(VK_CULL_MODE_NONE)
                    .frontFace(VK_FRONT_FACE_COUNTER_CLOCKWISE)
                    .lineWidth(1.0f);

            final VkPipelineMultisampleStateCreateInfo msState = VkPipelineMultisampleStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_MULTISAMPLE_STATE_CREATE_INFO)
                    .rasterizationSamples(v.msaaSamples == 0 ? VK_SAMPLE_COUNT_1_BIT : v.msaaSamples);

            // Alpha blending – preserve alpha in output buffer
            final VkPipelineColorBlendAttachmentState.Buffer colorAttachment =
                    VkPipelineColorBlendAttachmentState.calloc(1, stack)
                            .blendEnable(true)
                            .srcColorBlendFactor(VK_BLEND_FACTOR_SRC_ALPHA)
                            .dstColorBlendFactor(VK_BLEND_FACTOR_ONE_MINUS_SRC_ALPHA)
                            .colorBlendOp(VK_BLEND_OP_ADD)
                            .srcAlphaBlendFactor(VK_BLEND_FACTOR_ONE)
                            .dstAlphaBlendFactor(VK_BLEND_FACTOR_ONE_MINUS_SRC_ALPHA)
                            .alphaBlendOp(VK_BLEND_OP_ADD)
                            .colorWriteMask(
                                    VK_COLOR_COMPONENT_R_BIT | VK_COLOR_COMPONENT_G_BIT |
                                            VK_COLOR_COMPONENT_B_BIT | VK_COLOR_COMPONENT_A_BIT);

            final VkPipelineColorBlendStateCreateInfo blendState = VkPipelineColorBlendStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_COLOR_BLEND_STATE_CREATE_INFO)
                    .pAttachments(colorAttachment);

            final VkPipelineDepthStencilStateCreateInfo depthState = VkPipelineDepthStencilStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_DEPTH_STENCIL_STATE_CREATE_INFO);

            // Dynamic states: viewport + scissor
            final IntBuffer dynamicStates = stack.ints(VK_DYNAMIC_STATE_VIEWPORT, VK_DYNAMIC_STATE_SCISSOR);
            final VkPipelineDynamicStateCreateInfo dynState = VkPipelineDynamicStateCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_PIPELINE_DYNAMIC_STATE_CREATE_INFO)
                    .pDynamicStates(dynamicStates);

            final VkGraphicsPipelineCreateInfo.Buffer pipelineInfo = VkGraphicsPipelineCreateInfo.calloc(1, stack)
                    .sType(VK_STRUCTURE_TYPE_GRAPHICS_PIPELINE_CREATE_INFO)
                    .pStages(stages)
                    .pVertexInputState(vertexInput)
                    .pInputAssemblyState(inputAssembly)
                    .pViewportState(viewportState)
                    .pRasterizationState(rasterState)
                    .pMultisampleState(msState)
                    .pDepthStencilState(depthState)
                    .pColorBlendState(blendState)
                    .pDynamicState(dynState)
                    .layout(data.pipelineLayout)
                    .renderPass(v.renderPass)
                    .subpass(v.subpass);

            final LongBuffer pPipeline = stack.mallocLong(1);
            checkVk(vkCreateGraphicsPipelines(v.device, v.pipelineCache, pipelineInfo, v.allocator, pPipeline));
            return pPipeline.get(0);
        }
    }

    /**
     * Creates a {@code VkShaderModule} from a SPIR-V byte buffer.
     */
    private long createShaderModule(final ByteBuffer spirv) {
        final InitInfo v = data.initInfo;
        try (MemoryStack stack = MemoryStack.stackPush()) {
            final VkShaderModuleCreateInfo info = VkShaderModuleCreateInfo.calloc(stack)
                    .sType(VK_STRUCTURE_TYPE_SHADER_MODULE_CREATE_INFO)
                    .pCode(spirv);

            final LongBuffer pModule = stack.mallocLong(1);
            checkVk(vkCreateShaderModule(v.device, info, v.allocator, pModule));
            return pModule.get(0);
        }
    }

    /**
     * Finds a memory type index that satisfies {@code typeFilter} and {@code properties}.
     */
    private int findMemoryType(final int typeFilter, final int properties) {
        try (MemoryStack stack = MemoryStack.stackPush()) {
            final VkPhysicalDeviceMemoryProperties memProps = VkPhysicalDeviceMemoryProperties.malloc(stack);
            vkGetPhysicalDeviceMemoryProperties(data.initInfo.physicalDevice, memProps);
            for (int i = 0; i < memProps.memoryTypeCount(); i++) {
                if ((typeFilter & (1 << i)) != 0
                        && (memProps.memoryTypes(i).propertyFlags() & properties) == properties) {
                    return i;
                }
            }
        }
        throw new RuntimeException("ImGuiImplVk: failed to find suitable memory type");
    }

    /**
     * Calls the user-supplied result checker (if any).
     */
    private void checkVk(final int result) {
        if (data.initInfo.checkVkResult != null) {
            data.initInfo.checkVkResult.accept(result);
        }
    }

    /**
     * Functional interface for Vulkan result checking.
     */
    @FunctionalInterface
    public interface VkResultConsumer {
        void accept(int vkResult);
    }

    /**
     * Holds all Vulkan handles and settings required for initialisation.
     * Mirrors {@code ImGui_ImplVulkan_InitInfo} from the C++ backend.
     */
    public static final class InitInfo {
        /**
         * The Vulkan instance.
         */
        public VkInstance instance;
        /**
         * The physical device chosen for rendering.
         */
        public VkPhysicalDevice physicalDevice;
        /**
         * The logical device.
         */
        public VkDevice device;
        /**
         * The graphics queue family index.
         */
        public int queueFamily;
        /**
         * The graphics queue.
         */
        public VkQueue queue;
        /**
         * A descriptor pool with {@code VK_DESCRIPTOR_POOL_CREATE_FREE_DESCRIPTOR_SET_BIT}.
         * Must have room for at least {@value #MINIMUM_IMAGE_SAMPLER_POOL_SIZE} combined-image-sampler descriptors.
         */
